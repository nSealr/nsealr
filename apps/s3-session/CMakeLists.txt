cmake_minimum_required(VERSION 3.16)
set(COMPONENTS main)
set(nsealr_tinyusb "$ENV{NSEALR_S3_TINYUSB_COMPONENT}")
if(NOT EXISTS "${nsealr_tinyusb}/idf_component.yml")
    message(FATAL_ERROR "Provide the admitted esp_tinyusb 3.0.0 component")
endif()
file(SHA256 "${nsealr_tinyusb}/idf_component.yml" tinyusb_manifest_sha)
if(NOT tinyusb_manifest_sha STREQUAL "d3b0800b83099a4fb13139b463bebea93995593244daaf491fef75216bd80b4f")
    message(FATAL_ERROR "esp_tinyusb manifest differs from the admitted pin")
endif()
set(EXTRA_COMPONENT_DIRS "${nsealr_tinyusb}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(nsealr_s3_native_public)
