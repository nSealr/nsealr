cmake_minimum_required(VERSION 3.16)
if(NOT DEFINED NSEALR_P4_LOCK_FILE)
  message(FATAL_ERROR "Use build.py with explicit tool and work roots")
endif()
set(P4_WDT_PROBE_MODE "0" CACHE STRING "Public fixture: 0 normal, 1 busyCPU, 2 blockingwait, 3 public C6 reset")
if(NOT P4_WDT_PROBE_MODE MATCHES "^[0123]$")
  message(FATAL_ERROR "Invalid P4 public probe mode")
endif()
set(P4_ENTROPY_FIXED_UNAVAILABLE "0" CACHE STRING "Public entropy only: 0 real helper, 1 fixed Unavailable")
if(NOT P4_ENTROPY_FIXED_UNAVAILABLE MATCHES "^[01]$")
 message(FATAL_ERROR "Invalid P4 public entropy mode")
endif()
if(P4_ENTROPY_FIXED_UNAVAILABLE AND NOT P4_WDT_PROBE_MODE STREQUAL "0")
 message(FATAL_ERROR "Entropy negative requires public normal C6 guard mode")
endif()
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK "${NSEALR_P4_LOCK_FILE}")
project(nsealr_p4_public_optical)
