cmake_minimum_required(VERSION 3.18)
project(nsealr_ui_preview LANGUAGES C)
set(CMAKE_C_STANDARD 11)
if(NOT DEFINED LVGL_DIR OR NOT EXISTS "${LVGL_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Pass the pinned LVGL source directory")
endif()
if(NOT DEFINED NSEALR_CORE_ROOT)
  set(NSEALR_CORE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
endif()
if(NOT DEFINED RUST_LIBRARY OR NOT EXISTS "${RUST_LIBRARY}")
  message(FATAL_ERROR "Pass the current native nsealr-device static library")
endif()
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "" FORCE)
set(LV_FETCH_DEPENDENCIES OFF CACHE BOOL "" FORCE)
set(LV_USE_PKG_CONFIG OFF CACHE BOOL "" FORCE)
add_subdirectory("${LVGL_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/lvgl")
file(GLOB fonts "../ui/fonts/font_*.c")
add_executable(ui-preview preview.c ../device.c ../word_input.c ../backup_check.c ../ui/ui.c ../p4/main/secret_raster.c ../p4/main/secret_qr.c ../p4/main/board_rng_logic.c ${fonts})
target_include_directories(ui-preview PRIVATE .. ../p4/main "${NSEALR_CORE_ROOT}/core/nsealr-device/include" "${LVGL_DIR}")
target_compile_options(ui-preview PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation -UNDEBUG)
target_link_libraries(ui-preview PRIVATE lvgl "${RUST_LIBRARY}" m)
if(APPLE)
  target_link_libraries(ui-preview PRIVATE "-framework Security" "-framework SystemConfiguration")
else()
  target_link_libraries(ui-preview PRIVATE pthread dl)
endif()
enable_testing()
add_test(NAME device-flow COMMAND ui-preview --test)
